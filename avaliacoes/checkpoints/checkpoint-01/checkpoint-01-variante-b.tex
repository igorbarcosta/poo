\cabecalhoCenario{Elevador}
Ao criar um \codigoInline{Elevador}, \codigoInline{andarAtual} começa em \codigoInline{0} e
\codigoInline{portaAberta} em \codigoInline{false}. Considere exatamente os arquivos abaixo.

\noindent\begin{minipage}[t]{.37\textwidth}
\begin{painelreferencia}{Elevador.java}
\begin{lstlisting}[style=cenariocodigo]
public class Elevador {
    int andarAtual;
    boolean portaAberta;

    void subir() {
        andarAtual = andarAtual + 1;
    }
    void abrirPorta() {
        portaAberta = true;
    }
    void fecharPorta() {
        portaAberta = false;
    }
    int getAndarAtual() {
        return andarAtual;
    }
    boolean getPortaAberta() {
        return portaAberta;
    }
}
\end{lstlisting}
\end{painelreferencia}
\end{minipage}\hfill
\begin{minipage}[t]{.61\textwidth}
\begin{painelreferencia}{Main.java}
\begin{lstlisting}[style=cenariocodigo]
public class Main {
    public static void main(String[] args) {
        Elevador cabine = new Elevador();
        Elevador painel = cabine;
        Elevador carga = new Elevador();

        painel.subir();
        painel.abrirPorta();
        carga.subir();
        carga.subir();
        carga.fecharPorta();

        System.out.println(cabine == painel);
        System.out.println(cabine == carga);
        System.out.println(cabine.getAndarAtual());
        System.out.println(painel.getPortaAberta());
        System.out.println(carga.getAndarAtual());
        System.out.println(carga.getPortaAberta());
    }
}
\end{lstlisting}
\end{painelreferencia}
\end{minipage}

\questaoDireta{2}{Considere a evolução da classe e marque cada afirmação como V ou F.}{3,0 pontos}
Os métodos \codigoInline{abrirPorta()} e \codigoInline{fecharPorta()} permanecem públicos e inalterados.
\begin{painelEvolucao}{Trechos alterados em Elevador.java}
\begin{lstlisting}[style=evolucaocodigo]
private int andarAtual;
private boolean portaAberta;
public int getAndarAtual() { return andarAtual; }
public boolean getPortaAberta() { return portaAberta; }
public void subir() {
    if (portaAberta == false) {
        if (andarAtual < 3) {
            andarAtual = andarAtual + 1;
        }
    }
}
\end{lstlisting}
\end{painelEvolucao}
\itemVF{a}{Em \codigoInline{Main}, a instrução \codigoInline{elevador.portaAberta = true;} compila.}
\itemVF{b}{\codigoInline{Main} pode consultar a porta usando \codigoInline{getPortaAberta()}.}
\itemVF{c}{Do térreo, após \codigoInline{abrirPorta()} e \codigoInline{subir()}, o andar continua em \codigoInline{0}.}
\itemVF{d}{Com a porta fechada e o andar em \codigoInline{2}, chamar \codigoInline{subir()} leva ao andar \codigoInline{3}.}
\itemVF{e}{Métodos de \codigoInline{Elevador} não podem alterar \codigoInline{andarAtual} porque o campo é privado.}
\itemVF{f}{\codigoInline{Elevador painel = elevador;} copia a referência e não cria outro objeto.}

\questaoDireta{3}{Qual classificação está correta?}{0,5 ponto}
\begin{alternativas}
\item \codigoInline{subir()} é estado; \codigoInline{portaAberta} é comportamento.
\item Ambos são comportamento. \item Um é classe; o outro é objeto.
\item \codigoInline{portaAberta} é estado; \codigoInline{subir()} é comportamento. \item Ambos são estado.
\end{alternativas}
\questaoDireta{4}{O que \codigoInline{void} informa em \codigoInline{void fecharPorta()}?}{0,5 ponto}
\begin{alternativas}
\item O método não devolve valor ao chamador. \item O método não pode alterar campos.
\item Somente a classe pode chamar o método. \item O método sempre devolve \codigoInline{false}.
\item O método não possui parâmetros.
\end{alternativas}
\questaoDireta{5}{O que significa um método sem \codigoInline{public} ou \codigoInline{private}?}{0,5 ponto}
\begin{alternativas}
\item Recebe o acesso da classe. \item Possui acesso público. \item Sua declaração é inválida.
\item Possui acesso privado. \item Possui acesso de pacote.
\end{alternativas}
\questaoDireta{6}{Qual afirmação sobre uma variável \codigoInline{int} está correta?}{0,5 ponto}
\begin{alternativas}
\item Campo \codigoInline{int} começa em \codigoInline{false}.
\item Variável local começa em \codigoInline{0}; campo exige atribuição.
\item Campo começa em \codigoInline{0}; variável local exige atribuição antes do uso.
\item Campo e variável local começam em \codigoInline{0}. \item Nenhum deles pode usar \codigoInline{0}.
\end{alternativas}

\questaoDireta{7}{Qual opção mantém no elevador a responsabilidade pela porta?}{0,5 ponto}
\begin{alternativas}
\item \codigoInline{Main} escolhe e altera o campo.
\item \codigoInline{elevador.abrirPorta();}; o objeto realiza a mudança.
\item \codigoInline{Main} consulta e espera uma mudança.
\item Um método de \codigoInline{Main} altera o campo.
\item \codigoInline{Main} copia a referência para mudar a porta.
\end{alternativas}
\questaoDireta{8}{Qual afirmação distingue corretamente classe e objeto?}{0,5 ponto}
\begin{alternativas}
\item Atribuir uma referência cria outro objeto. \item Cada variável declarada cria um objeto.
\item Objetos da mesma classe têm sempre o mesmo estado.
\item A classe define elementos comuns; cada objeto mantém seu estado.
\item \codigoInline{new Elevador()} cria uma nova classe.
\end{alternativas}
\questaoDireta{9}{O que ocorre ao executar \codigoInline{new Elevador()}?}{0,5 ponto}
\begin{alternativas}
\item Um novo objeto \codigoInline{Elevador} é criado. \item A classe é declarada novamente.
\item Todos os elevadores passam a compartilhar estado. \item Um método passa a ser público.
\item Uma variável existente muda de nome.
\end{alternativas}
\questaoDireta{10}{Por que \codigoInline{getPortaAberta()} acessa \codigoInline{portaAberta} sem recebê-la como parâmetro?}{0,5 ponto}
\begin{alternativas}
\item \codigoInline{boolean} não pode ser parâmetro.
\item Todo método recebe os campos automaticamente como parâmetros.
\item O método usa o estado do próprio objeto.
\item Um getter cria outro objeto para guardar o valor.
\item \codigoInline{public} transforma o campo em variável local.
\end{alternativas}
